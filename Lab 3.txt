2.Develop an Event Logging System that records login, logout, failed-login, file-upload, and password-change events. Use MongoDB queries to identify repeated failed-login attempts and frequently occurring events. 
  use EventLogDB

  // Insert sample event logs
  db.logs.insertMany([
  { username: "user1", eventType: "failed-login", ip: "192.168.1.10", timestamp: new Date() },
  { username: "user1", eventType: "failed-login", ip: "192.168.1.10", timestamp: new Date() },
  { username: "user1", eventType: "failed-login", ip: "192.168.1.10", timestamp: new Date() },
  { username: "admin", eventType: "login", ip: "192.168.1.5", timestamp: new Date() },
  { username: "user2", eventType: "file-upload", ip: "192.168.1.20", timestamp: new Date() }
  ]);

  // Query: Identify repeated failed-login attempts (> 2 times)
  db.logs.aggregate([
  { $match: { eventType: "failed-login" } },
  { $group: { _id: "$username", failedAttempts: { $sum: 1 } } },
  { $match: { failedAttempts: { $gt: 2 } } }
  ]);

  // Query: Find frequently occurring event types
  db.logs.aggregate([
  { $group: { _id: "$eventType", totalCount: { $sum: 1 } } },
  { $sort: { totalCount: -1 } }
  ]);

  {

8.Implement a Shopping Cart System using embedded MongoDB documents. Perform operations for adding products, changing quantities, removing products, and calculating the cart value.
  use ShoppingCartDB

  // Insert cart document for a user
  db.carts.insertOne({
  userId: "U101",
  items: [
    { productId: "P01", name: "Wireless Mouse", qty: 1, price: 800 },
    { productId: "P02", name: "USB Hub", qty: 2, price: 400 }
  ]
  });

  // Operation: Add product / change quantity (Increment USB Hub quantity by 1)
  db.carts.updateOne(
  { userId: "U101", "items.productId": "P02" },
  { $inc: { "items.$.qty": 1 } }
  );

  // Operation: Calculate cart total value
  db.carts.aggregate([
  { $match: { userId: "U101" } },
  { $unwind: "$items" },
  { $group: { _id: "$userId", cartTotal: { $sum: { $multiply: ["$items.qty", "$items.price"] } } } }
  ]);

11.Implement a Bank Wallet Transfer System using MongoDB transactions. Transfer an amount between two users and ensure that both balance updates are completed atomically.
  use BankDB

  db.accounts.insertMany([
  { accountNumber: "ACC1001", holder: "Aarav", balance: 5000 },
  { accountNumber: "ACC1002", holder: "Priya", balance: 3000 }
  ]);

  // Execute atomic transaction for money transfer
  const session = db.getMongo().startSession();
  session.startTransaction();
  try {
  const accountsCollection = session.getDatabase("BankDB").accounts;
  
  // Deduct from sender
  accountsCollection.updateOne({ accountNumber: "ACC1001" }, { $inc: { balance: -1000 } }, { session });
  
  // Add to receiver
  accountsCollection.updateOne({ accountNumber: "ACC1002" }, { $inc: { balance: 1000 } }, { session });
  
  session.commitTransaction();
  print("Transaction committed successfully.");
  } catch (error) {
  session.abortTransaction();
  print("Transaction aborted due to error: " + error);
  } finally {
  session.endSession();
  }
