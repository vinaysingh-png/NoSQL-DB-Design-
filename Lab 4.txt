2.Using Apache HBase, perform basic CRUD operations on an Employee table. Demonstrate create, put, get, scan, and delete operations.
  # Create table
  create 'Employee', 'info', 'salary'

  # 1. Put (Create/Insert record)
  put 'Employee', 'E01', 'info:name', 'Rohan'
  put 'Employee', 'E01', 'info:dept', 'IT'
  put 'Employee', 'E01', 'salary:basic', '50000'

  # 2. Get (Retrieve specific record)
  get 'Employee', 'E01'

  # 3. Scan (Retrieve all records)
  scan 'Employee'

  # 4. Update (Overwrite value using put)
  put 'Employee', 'E01', 'salary:basic', '55000'

  # 5. Delete operations
  delete 'Employee', 'E01', 'info:dept'
  deleteall 'Employee', 'E01'
