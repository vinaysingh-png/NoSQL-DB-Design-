  // 4. Calculate total order amount using aggregation
  db.Orders.aggregate([

  { $group: { _id: "$orderId", totalAmount: { $sum: { $multiply: ["$products.qty", "$products.price"] } } } }
  ])

7. Neo4j Graph Database (Cypher Queries)
  // 1. Create students and courses, and ENROLLED_IN relationships
  CREATE (s1:Student {name: 'Aarav', id: 1})
  CREATE (s2:Student {name: 'Priya', id: 2})
  CREATE (s3:Student {name: 'Rahul', id: 3})
  CREATE (s4:Student {name: 'Neha', id: 4})

  CREATE (c1:Course {title: 'NoSQL Databases'})
  CREATE (c2:Course {title: 'Data Analytics'})
  CREATE (c3:Course {title: 'Web Development'})

  CREATE (s1)-[:ENROLLED_IN]->(c1)
  CREATE (s1)-[:ENROLLED_IN]->(c2)
  CREATE (s2)-[:ENROLLED_IN]->(c1)
  CREATE (s3)-[:ENROLLED_IN]->(c3)
  CREATE (s4)-[:ENROLLED_IN]->(c2)
  CREATE (s5)-[:ENROLLED_IN]->(c1);

  // 2. Display all students
  MATCH (s:Student) RETURN s;

  // 3. Find students enrolled in a particular course
  MATCH (s:Student)-[:ENROLLED_IN]->(c:Course {title: 'NoSQL Databases'}) RETURN s;

  // 4. Display all courses taken by a particular student
  MATCH (s:Student {name: 'Aarav'})-[:ENROLLED_IN]->(c:Course) RETURN c;

  // 5. Add and remove a relationship
  // Add relationship:
  MATCH (s:Student {name: 'Priya'}), (c:Course {title: 'Data Analytics'})
  CREATE (s)-[:ENROLLED_IN]->(c);

  // Remove relationship:
  MATCH (s:Student {name: 'Aarav'})-[r:ENROLLED_IN]->(c:Course {title: 'Data Analytics'})
  DELETE r;

14. MapReduce Operation in MongoDB
  // Map Function: Emits course as key and 1 (for count) or marks (for average) as value
  var mapFunction = function() {
    emit(this.Course, { count: 1, totalMarks: this.Marks });
  };

  // Reduce Function: Aggregates counts and total marks per course
  var reduceFunction = function(key, values) {
    reducedVal = { count: 0, totalMarks: 0 };
    for (var i = 0; i < values.length; i++) {
        reducedVal.count += values[i].count;
        reducedVal.totalMarks += values[i].totalMarks;
    }
    return reducedVal;
  };

  // Finalize Function: Computes average marks
  var finalizeFunction = function(key, reducedVal) {
    reducedVal.avgMarks = reducedVal.totalMarks / reducedVal.count;
    return reducedVal;
  };

  // Execute MapReduce
  db.Students.mapReduce(
    mapFunction,
    reduceFunction,
    {
        out: "course_summary",
        finalize: finalizeFunction
    }
  );

  // View Results
  db.course_summary.find().pretty();
